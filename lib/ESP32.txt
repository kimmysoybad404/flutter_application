#include <Firebase_ESP_Client.h>
#include <DHT.h>
#include <NTPClient.h>
#include <WiFiUdp.h>

// -------------------- WiFi --------------------
#define WIFI_SSID "Kimmy"
#define WIFI_PASSWORD "kimmy1234"

// -------------------- Firebase --------------------
#define API_KEY "AIzaSyBOOj-0XLlpUFdqdWFB6Fy2zQ75U54vRaU"
#define DATABASE_URL "smarthomea-43c25-default-rtdb.asia-southeast1.firebasedatabase.app"

FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// -------------------- DHT22 Sensor --------------------
#define DHTPIN 4
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

// -------------------- NTP (เวลา) --------------------
WiFiUDP ntpUDP;
NTPClient timeClient(ntpUDP, "pool.ntp.org", 7 * 3600); // GMT+7


void setup() {
  Serial.begin(115200);

  // -------------------- WiFi --------------------
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n✅ WiFi connected");

  // -------------------- Firebase --------------------
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  // anonymous login
  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("✅ Firebase SignUp OK");
  } else {
    Serial.printf("❌ Firebase SignUp failed: %s\n",
                  config.signer.signupError.message.c_str());
  }

  // ตั้งค่า timeout กัน SSL ปิดเอง
  config.timeout.serverResponse = 10 * 1000;      // 10s
  config.timeout.socketConnection = 10 * 1000;
  config.timeout.sslHandshake = 10 * 1000;
  config.timeout.rtdbKeepAlive = 45 * 1000;       // keepAlive

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  dht.begin();
  Serial.println("✅ DHT22 sensor ready");
}

void loop() {
  timeClient.update();

  float temp = dht.readTemperature();

  if (!isnan(temp)) {
    // วันที่แบบ YYYY-MM-DD
    String temoTxt = String(temp) + " °C"
    time_t rawTime = timeClient.getEpochTime();
    struct tm *ti = localtime(&rawTime);
    char dateStr[20];
    sprintf(dateStr, "%04d-%02d-%02d", ti->tm_year + 1900, ti->tm_mon + 1, ti->tm_mday);

    // เวลาแบบ HH:MM:SS
    String timeStr = timeClient.getFormattedTime();

    Serial.printf("🌡 Temp: %.2f °C | 📅 %s 🕒 %s\n", temp, dateStr, timeStr.c_str());

    if (Firebase.ready()) {
      Serial.println("🔥 Firebase is ready, sending...");

      // Path: /weather/วันที่/เวลา
      String path = "/weather/" + String(dateStr) + "/" + timeStr;

      if (Firebase.RTDB.setFloat(&fbdo, path.c_str(), temp)) {
        Serial.println("✅ Saved to Firebase");
      } else {
        Serial.printf("❌ Firebase error: %s\n", fbdo.errorReason().c_str());
      }
    }
  } else {
    Serial.println("❌ Failed to read DHT22");
  }

  delay(10000); // ส่งทุก 10 วิ
}
