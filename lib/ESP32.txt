#include <WiFi.h>
#include <Firebase_ESP_Client.h>
#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <DHT.h>

// -------------------- WiFi --------------------
#define WIFI_SSID "Kimmy"
#define WIFI_PASSWORD "kimmy1234"

// -------------------- Firebase --------------------
#define API_KEY "AIzaSyBOOj-0XLlpUFdqdWFB6Fy2zQ75U54vRaU"
#define DATABASE_URL "https://smarthomea-43c25-default-rtdb.asia-southeast1.firebasedatabase.app/"

// Firebase objects
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// -------------------- Sensors --------------------
#define DHTPIN 4
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("🚀 ESP32 Starting...");

  // -------------------- WiFi --------------------
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("🔌 Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.print("\n✅ WiFi connected, IP: ");
  Serial.println(WiFi.localIP());

  // -------------------- Firebase --------------------
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  // ถ้าไม่ได้ใช้ Auth ให้ปล่อย email/password ว่างไว้
  auth.user.email = "";
  auth.user.password = "";

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  Serial.println("✅ Firebase initialized");

  // -------------------- DHT22 --------------------
  dht.begin();
  Serial.println("✅ DHT22 sensor ready");
}

void loop() {
  Serial.println("==============================");

  // อ่านค่า DHT22
  float temperatureDHT = dht.readTemperature();

  if (isnan(temperatureDHT)) {
    Serial.println("❌ Failed to read DHT22");
  } else {
    Serial.printf("🌡 Temperature: %.2f °C\n", temperatureDHT);

    // -------------------- ส่งขึ้น Firebase --------------------
    if (Firebase.ready()) {
      Serial.println("🔥 Firebase is ready, sending data...");
      if (Firebase.RTDB.setFloat(&fbdo, "/weather/temperature", temperatureDHT)) {
        Serial.println("✅ Temperature sent to Firebase!");
      } else {
        Serial.print("❌ Firebase error: ");
        Serial.println(fbdo.errorReason());
      }
    } else {
      Serial.println("⚠️ Firebase not ready");
    }
  }

  Serial.println("==============================\n");
  delay(5000);
}
